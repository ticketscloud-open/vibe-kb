[[Vibe-Admin Customers|Админка пользователей]]

```typescript

interface Customer {
	id: number 				// Идентификатор пользователя в Vibe
	created: number 		// Дата и время создания записи
	updated: number 		// Дата и время обновления записи
	is_active: boolean 		// Пользователь активный (не отключён) 
	authorized: number 		// Дата и время последней авторизации
	phone: string 			// Номер телефона
	email: string			// Emal
	bio: string				// О себе
	language: string		// Установленный язык
	first_name: string		// Имя
	last_name: string		// Фамилия
	nickname: string		// Никнейм в сервисе
	avatar_uri: string  	// URL аватара
	birthday: number    	// Дата и время рождения
	gender: "male" | "female" | "other" // Пол
	notify_cash: boolean	// Настройки оповещений (финансы)
	notify_chat: boolean	// Настройки оповещений (чаты, знакомства)
	notify_award: boolean	// Настройки оповещений (вознаграждения)
	notify_ads: boolean		// Настройки оповещений (реклама)
	is_super: boolean		// Доступ в админку Vibe
	meta: object			// Прочие schemaless данные
}
```
