[[Vibe-Admin Customers|Админка пользователей]]

```typescript

interface Customer {
	// Идентификатор пользователя в Vibe
	id: number
	// Дата и время создания записи
	created: number
	// Дата и время обновления записи 		
	updated: number 		
	// Дата и время последней авторизации
	authorized: number
	// При деактивации пользователь будет отключен от сервиса
	is_active: boolean
	// Номер телефона
	phone: string 			
	// Email
	email: string
	// О себе
	bio: string				
	// Установленный язык
	language: string		
	// Имя
	first_name: string		
	// Фамилия
	last_name: string		
	// Никнейм в сервисе
	nickname: string		
	// URL аватара
	avatar_uri: string  	
	// Дата и время рождения
	birthday: number    	
	// Пол
	gender: "male" | "female" | "other" 
	// Настройки оповещений (финансы)
	notify_cash: boolean	
	// Настройки оповещений (чаты, знакомства)
	notify_chat: boolean	
	// Настройки оповещений (вознаграждения)
	notify_award: boolean	
	// Настройки оповещений (реклама)
	notify_ads: boolean		
	// Доступ в админку Vibe
	is_super: boolean		
	// Прочие schemaless данные
	meta: object			
}
```
